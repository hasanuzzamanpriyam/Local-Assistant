# Phase 2 — Computer Control Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add computer control to the assistant — Python sidecar that observes screen, clicks, types, scrolls, and runs an agent loop, bridged to the Tauri frontend via NDJSON.

**Architecture:** Tauri manages a Python sidecar process (embedded venv via `uv`). NDJSON over stdin/stdout for IPC. Agent loop: observe (screenshot + OCR) → plan (LLM) → act (pyautogui) → repeat.

**Tech Stack:** Rust (Tauri commands + sidecar manager), Python 3.11 (mss, pyautogui, pywin32, pytesseract, requests, Pillow), Svelte 5 (frontend components/stores), uv (Python venv manager)

## Global Constraints

- Target hardware: HP EliteBook 845 G8 — Ryzen 5000, 16 GB RAM, Windows 11
- All-local, free, no cloud dependencies
- Python sidecar: embedded venv via `uv`, no system Python required
- NDJSON protocol: exact message types from spec §3
- Native commands: exact signatures from spec §4
- Default max_steps: 10
- Screenshots captured on every agent step
- Sidecar auto-respawn (max 3) on crash

def decode(line: str) -> Optional[dict[str, Any]]:
    line = line.strip()
    if not line:
        return None
    try:
        return json.loads(line)
    except json.JSONDecodeError:
        return None

import base64
import io
from mss import mss


def list_monitors() -> list[dict]:
    with mss() as sct:
        return [
            {"id": i, "x": m["left"], "y": m["top"], "width": m["width"], "height": m["height"]}
            for i, m in enumerate(sct.monitors[1:], start=1)
        ]


def capture() -> str:
    with mss() as sct:
        monitor = sct.monitors[0]
        img = sct.grab(monitor)
        buf = io.BytesIO()
        from PIL import Image
        Image.frombytes("RGB", img.size, img.rgb).save(buf, format="PNG")
        return base64.b64encode(buf.getvalue()).decode("ascii")


def capture_region(x: int, y: int, w: int, h: int) -> str:
    with mss() as sct:
        region = {"top": y, "left": x, "width": w, "height": h}
        img = sct.grab(region)
        buf = io.BytesIO()
        from PIL import Image
        Image.frombytes("RGB", img.size, img.rgb).save(buf, format="PNG")
        return base64.b64encode(buf.getvalue()).decode("ascii")

import base64
import io
from PIL import Image

try:
    import pytesseract
    TESSERACT_AVAILABLE = True
except Exception:
    TESSERACT_AVAILABLE = False


def _decode_image(image_b64: str) -> Image.Image:
    img_data = base64.b64decode(image_b64)
    return Image.open(io.BytesIO(img_data))


def extract_text(image_b64: str) -> list[dict]:
    if not TESSERACT_AVAILABLE:
        return []
    img = _decode_image(image_b64)
    data = pytesseract.image_to_data(img, output_type=pytesseract.Output.DICT)
    results = []
    for i in range(len(data["text"])):
        text = data["text"][i].strip()
        if text:
            results.append({
                "text": text,
                "x": data["left"][i],
                "y": data["top"][i],
                "width": data["width"][i],
                "height": data["height"][i],
                "confidence": data["conf"][i],
            })
    return results


def find_text(image_b64: str, query: str) -> list[dict]:
    if not TESSERACT_AVAILABLE:
        return []
    all_text = extract_text(image_b64)
    query_lower = query.lower()
    return [item for item in all_text if query_lower in item["text"].lower()]
```

import time
from screen import capture
from ocr import extract_text
from actions import screenshot as action_screenshot, wait, click_element, type_into, scroll_page, hotkey_combo
from protocol import MSG_STEP, MSG_TASK_COMPLETE, MSG_ERROR


class AgentLoop:
    def __init__(self, emit=None, max_steps=10):
        self.emit = emit
        self.max_steps = max_steps
        self.llm_url = None

    def run(self, task: str, max_steps: int, llm_url: str) -> dict:
        self.llm_url = llm_url
        self.max_steps = max_steps
        start = time.time()
        steps = 0

        if self.emit:
            self.emit({"event": "task_started", "task": task, "max_steps": max_steps})

        history = []
        for step in range(1, max_steps + 1):
            steps = step

            # Observe
            img_b64 = capture()
            text_items = extract_text(img_b64)
            ocr_text = "\n".join(item["text"] for item in text_items)

            # Plan (call LLM)
            action = self._plan(task, img_b64, ocr_text, history)

            if self.emit:
                self.emit({
                    "event": MSG_STEP,
                    "step": step,
                    "max_steps": max_steps,
                    "thought": action.get("thought", ""),
                    "action": action.get("action", ""),
                    "screenshot_base64": img_b64,
                })

            history.append(action)

            # Act
            if action.get("action") == "done":
                duration_ms = int((time.time() - start) * 1000)
                if self.emit:
                    self.emit({
                        "event": MSG_TASK_COMPLETE,
                        "success": True,
                        "result": action.get("result", "Task complete"),
                        "steps": steps,
                        "duration_ms": duration_ms,
                    })
                return {"success": True, "steps": steps, "duration_ms": duration_ms}

            self._execute_action(action)

        # Max steps reached
        duration_ms = int((time.time() - start) * 1000)
        if self.emit:
            self.emit({
                "event": MSG_TASK_COMPLETE,
                "success": False,
                "result": f"Max steps ({max_steps}) reached",
                "steps": steps,
                "duration_ms": duration_ms,
            })
        return {"success": False, "steps": steps, "duration_ms": duration_ms}

    def _plan(self, task: str, img_b64: str, ocr_text: str, history: list) -> dict:
        import requests
        prompt = f"""Task: {task}
OCR Text: {ocr_text}
History: {history}
Decide the next action. Respond with JSON: {{"thought": "...", "action": "click|type|scroll|wait|hotkey|done", "target": "...", "text": "...", "result": "..."}}
"""
        try:
            resp = requests.post(
                f"{self.llm_url}/v1/chat/completions",
                json={"model": "default", "messages": [{"role": "user", "content": prompt}], "stream": False},
                timeout=30,
            )
            content = resp.json()["choices"][0]["message"]["content"]
            import json as _json
            return _json.loads(content)
        except Exception:
            return {"thought": "LLM unavailable", "action": "done", "result": "LLM call failed"}

    def _execute_action(self, action: dict) -> None:
        act = action.get("action", "")
        if act == "click":
            target = action.get("target", "")
            if target:
                click_element(target)
        elif act == "type":
            type_into(action.get("text", ""))
        elif act == "scroll":
            scroll_page(action.get("direction", "down"))
        elif act == "wait":
            wait(action.get("seconds", 1.0))
        elif act == "hotkey":
            hotkey_combo(action.get("keys", []))
```

import sys
import json
import threading
from protocol import encode, decode, CMD_RUN, CMD_STOP, CMD_PING, CMD_SHUTDOWN, MSG_READY, MSG_BOOTING, MSG_LOAD_ERROR, MSG_ERROR
from agent import AgentLoop


class Runner:
    def __init__(self):
        self.tasks = {}
        self.running = False
        self.llm_url = "http://127.0.0.1:11434/v1"  # default, overridden by Tauri

    def start(self):
        self.running = True
        # Emit booting
        print(encode({"event": "booting", "pid": __import__("os").getpid()}), flush=True)
        # Emit ready
        print(encode({"event": "ready"}), flush=True)

        # NDJSON stdin loop
        for line in sys.stdin:
            msg = decode(line)
            if msg is None:
                continue
            response = self.handle_command(msg)
            if response:
                print(encode(response), flush=True)
            if msg.get("cmd") == CMD_SHUTDOWN:
                break

    def handle_command(self, cmd: dict) -> dict | None:
        c = cmd.get("cmd")
        if c == CMD_PING:
            return {"cmd": "pong"}
        elif c == CMD_RUN:
            task_id = cmd.get("id")
            task = cmd.get("task")
            max_steps = cmd.get("max_steps", 10)
            agent = AgentLoop(emit=self._make_emit(task_id))
            thread = threading.Thread(target=agent.run, args=(task, max_steps, self.llm_url), daemon=True)
            self.tasks[task_id] = {"agent": agent, "thread": thread}
            thread.start()
            return {"event": "task_started", "id": task_id}
        elif c == CMD_STOP:
            task_id = cmd.get("id")
            if task_id in self.tasks:
                del self.tasks[task_id]
            return {"event": "task_stopped", "id": task_id}
        elif c == CMD_SHUTDOWN:
            self.running = False
            return {"event": "shutdown"}
        return None

    def _make_emit(self, task_id: str):
        def emit(event: dict):
            event["id"] = task_id
            print(encode(event), flush=True)
        return emit


if __name__ == "__main__":
    Runner().start()

use std::io::{Write, BufRead, BufReader};
use std::sync::{Arc, Mutex};
use std::thread;
use serde::{Deserialize, Serialize};
use anyhow::Result;
use tauri::Emitter;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SidecarStatus {
    pub running: bool,
    pub ready: bool,
    pub pid: Option<u32>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SidecarEvent {
    pub event: String,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub id: Option<String>,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub step: Option<u32>,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub max_steps: Option<u32>,

    pub success: Option<bool>,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub steps: Option<u32>,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub duration_ms: Option<u64>,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub error: Option<String>,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub fatal: Option<bool>,
}

pub struct SidecarManager {
    child: Mutex<Option<Child>>,
    stdin: Mutex<Option<std::process::ChildStdin>>,
    status: Mutex<SidecarStatus>,
    respawn_count: Mutex<u32>,
}

impl SidecarManager {
    pub fn new() -> Self {
        Self {
            child: Mutex::new(None),
            stdin: Mutex::new(None),
            status: Mutex::new(SidecarStatus { running: false, ready: false, pid: None }),
            respawn_count: Mutex::new(0),
        }
    }

    pub fn is_running(&self) -> bool {
        self.status.lock().unwrap().running
    }

    pub fn status(&self) -> SidecarStatus {
        self.status.lock().unwrap().clone()
    }

    pub fn start(&self) -> Result<()> {
        let mut status = self.status.lock().unwrap();
        if status.running {
            return Ok(());
        }
        drop(status);

        let mut cmd = Command::new("python")
            .arg("-u")
            .arg("sidecar/runner.py")
            .stdout(Stdio::piped())
            .stderr(Stdio::piped())
            .stdin(Stdio::piped())
            .spawn()?;

        let pid = cmd.id();
        let stdin = cmd.stdin.take().unwrap();
        let stdout = cmd.stdout.take().unwrap();

        *self.child.lock().unwrap() = Some(cmd);
        *self.stdin.lock().unwrap() = Some(stdin);

        *self.status.lock().unwrap() = SidecarStatus { running: true, ready: false, pid };

        // Spawn reader thread for NDJSON stdout
        let reader = BufReader::new(stdout);
        let status_clone = Arc::new(Mutex::new(self.status.lock().unwrap().clone()));
        let respawn_count = Arc::new(Mutex::new(self.respawn_count.lock().unwrap().clone()));

        thread::spawn(move || {
            for line in reader.lines() {
                match line {
                    Ok(l) => {
                        // Parse NDJSON and update status
                        if let Ok(event) = serde_json::from_str::<serde_json::Value>(&l) {
                            if let Some(evt) = event.get("event").and_then(|v| v.as_str()) {
                                match evt {
                                    "ready" => {
                                        let mut s = status_clone.lock().unwrap();
                                        s.ready = true;
                                    }
                                    "load_error" => {
                                        let mut s = status_clone.lock().unwrap();
                                        s.running = false;
                                    }
                                    _ => {}
                                }
                            }
                        }
                    }
                    Err(_) => {
                        // Sidecar crashed — attempt respawn
                        let mut rc = respawn_count.lock().unwrap();
                        if *rc < 3 {
                            *rc += 1;
                            drop(rc);
                            // Note: full respawn logic would need manager reference
                        }
                    }
                }
            }
        });

        Ok(())
    }

    pub fn stop(&self) -> Result<()> {
        let mut child = self.child.lock().unwrap();
        if let Some(mut c) = child.take() {
            c.kill()?;
            c.wait()?;
        }
        *self.stdin.lock().unwrap() = None;
        *self.status.lock().unwrap() = SidecarStatus { running: false, ready: false, pid: None };
        Ok(())
    }

    pub fn send_command(&self, command: serde_json::Value) -> Result<()> {
        let stdin = self.stdin.lock().unwrap();
        if let Some(ref mut s) = *stdin {
            writeln!(s, "{}", serde_json::to_string(&command)?)?;
        }
        Ok(())
    }
}

impl Default for SidecarManager {
    fn default() -> Self {
        Self::new()
    }
}

export interface SidecarStatus {
  running: boolean;
  ready: boolean;
  pid?: number;
}

export interface TaskEvent {
  id: string;
  event: string;
  step?: number;
  max_steps?: number;
  thought?: string;
  action?: string;
  screenshot_base64?: string;
  result?: string;
  success?: boolean;
  steps?: number;
  duration_ms?: number;
  error?: string;
}

  id: string;
  task: string;
  max_steps: number;
  status: 'running' | 'completed' | 'failed' | 'stopped';
  events: TaskEvent[];
}

const sidecarStatus = writable<SidecarStatus>({ running: false, ready: false });
const tasks = writable<ActiveTask[]>([]);
const events = writable<TaskEvent[]>([]);

function createSidecarStore() {
  return {
    subscribe: sidecarStatus.subscribe,
    subscribeTasks: tasks.subscribe,
    subscribeEvents: events.subscribe,

    async status(): Promise<SidecarStatus> {
      return invoke('sidecar_status');
    },

    async start(): Promise<void> {
      await invoke('sidecar_start');
    },

    async stop(): Promise<void> {
      await invoke('sidecar_stop');
    },

    async runTask(task: string, maxSteps = 10): Promise<string> {
      const result: { task_id: string } = await invoke('sidecar_run_task', { task, max_steps: maxSteps });
      return result.task_id;
    },

    async stopTask(taskId: string): Promise<void> {
      await invoke('sidecar_stop_task', { task_id: taskId });
    },

    async sendCommand(command: Record<string, any>): Promise<void> {
      await invoke('sidecar_send_command', { command });
    },

    // Listen for sidecar events via Tauri event system
    listen(callback: (event: TaskEvent) => void): Promise<() => void> {
      const unlisten = await window.__TAURI__.event.listen('sidecar-event', (e) => {
        callback(e.payload as TaskEvent);
      });
      return unlisten;
    },
  };
}

export const sidecarStore = createSidecarStore();

  let isLoading = $state(true);
  let showSettings = $state(false);
  let showSidecar = $state(false);

  onMount(async () => {
    await checkModel();
  });

  async function checkModel() {
    isLoading = true;
    isModelReady = false;
    try {
      const health = await modelHealth();
      if (health.healthy) {
        isModelReady = true;
      } else {
        await modelStart();
        await new Promise(r => setTimeout(r, 2000));
        const health2 = await modelHealth();
        isModelReady = health2.healthy;
      }
    } catch (error) {
      console.error('Model check failed:', error);
      isModelReady = false;
    } finally {
      isLoading = false;
    }
  }

  function openSettings() { showSettings = true; }
  function closeSettings() { showSettings = false; }
